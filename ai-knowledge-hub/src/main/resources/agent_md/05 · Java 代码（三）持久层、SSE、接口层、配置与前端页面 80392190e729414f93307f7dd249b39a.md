```sql
CREATE TABLE IF NOT EXISTS agent_session (

-- ============ 工具调用审计 ============
CREATE TABLE IF NOT EXISTS agent_tool_call (
  id          BIGINT       NOT NULL AUTO_INCREMENT,
  session_id  VARCHAR(64)  NOT NULL,
  user_id     VARCHAR(64)  DEFAULT NULL,

  args_json   LONGTEXT     DEFAULT NULL,

  KEY idx_audit_tool (tool_name, created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='工具调用审计';

-- ============ 仿真任务（建议对接你现有表，这里给最小集） ============
CREATE TABLE IF NOT EXISTS sim_task (
  id            BIGINT       NOT NULL AUTO_INCREMENT,
  session_id    VARCHAR(64)  DEFAULT NULL COMMENT '来自 Agent 时写入，手工提交为 NULL',
  user_id       VARCHAR(64)  NOT NULL,
  source        VARCHAR(16)  NOT NULL DEFAULT 'AGENT' COMMENT 'AGENT|FORM',
  plugin        VARCHAR(64)  NOT NULL COMMENT 'coventor|litho',
  workflow_id   VARCHAR(128) NOT NULL,
  params_json   LONGTEXT     DEFAULT NULL,
  mcp_job_id    VARCHAR(64)  DEFAULT NULL,
  external_ref  VARCHAR(64)  DEFAULT NULL,
  status        VARCHAR(24)  NOT NULL DEFAULT 'CREATED'
```

## 3. `conversation/ConversationStore.java` —— 上下文持久化

<aside>
⚠️

`loadMessages` 里的**孤立 tool 消息修剪**是个必要的坑：OpenAI 要求 `role=tool` 的消息必须紧跟在带 `tool_calls` 的 assistant 消息之后。滑窗切断时如果窗口头部是 tool 消息，请求会直接 400。

</aside>

```java
package com.acme.simagent.conversation;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;
import java.util.Map;
import java.util.UUID;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Repository;

import com.acme.simagent.llm.ChatMessage;
import com.acme.simagent.llm.ToolCall;
import com.fasterxml.jackson.databind.ObjectMapper;

/** 会话、消息、挂起动作、审计的落库（只用 JdbcTemplate，无 ORM 依赖）。 */
@Repository
public class ConversationStore {

    private static final Logger log = LoggerFactory.getLogger(ConversationStore.class);

    private final JdbcTemplate jdbc;
    private final ObjectMapper mapper = new ObjectMapper();

    public ConversationStore(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    // ---------------------------------------------------------- 会话

    public String createSession(String userId, String title) {
        String id = UUID.randomUUID().toString().replace("-", "");
        jdbc.update("INSERT INTO agent_session (id, user_id, title) VALUES (?, ?, ?)",
                id, userId, title == null ? "新会话" : trim(title, 200));
        return id;
    }

    public boolean sessionExists(String sessionId) {
        Integer n = jdbc.queryForObject(
                "SELECT COUNT(1) FROM agent_session WHERE id = ?", Integer.class, sessionId);
        return n != null && n > 0;
    }

    /** 不存在则创建，保证前端传任意 sessionId 都能用。 */
    public String ensureSession(String sessionId, String userId, String title) {
        if (sessionId == null || sessionId.isEmpty()) {
            return createSession(userId, title);
        }
        if (!sessionExists(sessionId)) {
            jdbc.update("INSERT INTO agent_session (id, user_id, title) VALUES (?, ?, ?)",
                    sessionId, userId, title == null ? "新会话" : trim(title, 200));
        }
        return sessionId;
    }

    // ---------------------------------------------------------- 消息

    public void appendMessage(String sessionId, ChatMessage msg) {
        String toolCallsJson = null;
        if (msg.hasToolCalls()) {
            try {
                toolCallsJson = mapper.writeValueAsString(msg.getToolCalls());
            } catch (Exception ex) {
                log.warn("序列化 tool_calls 失败", ex);
            }
        }
        jdbc.update("INSERT INTO agent_message "
                        + "(session_id, role, content, tool_calls_json, tool_call_id) "
                        + "VALUES (?, ?, ?, ?, ?)",
                sessionId, msg.getRole(), msg.getContent(), toolCallsJson, msg.getToolCallId());
        jdbc.update("UPDATE agent_session SET updated_at = NOW() WHERE id = ?", sessionId);
    }

    /** 取最近 limit 条，并修剪掉开头的孤立 tool 消息。 */
    public List<ChatMessage> loadMessages(String sessionId, int limit) {
        List<Map<String, Object>> rows = jdbc.queryForList(
                "SELECT role, content, tool_calls_json, tool_call_id FROM agent_message "
                        + "WHERE session_id = ? ORDER BY id DESC LIMIT ?", sessionId, limit);
        Collections.reverse(rows);

        List<ChatMessage> messages = new ArrayList<ChatMessage>();
        for (Map<String, Object> row : rows) {
            String role = (String) row.get("role");
            String content = (String) row.get("content");
            String toolCallsJson = (String) row.get("tool_calls_json");
            String toolCallId = (String) row.get("tool_call_id");

            if ("tool".equals(role)) {
                messages.add(ChatMessage.tool(toolCallId, content));
            } else if ("assistant".equals(role) && toolCallsJson != null) {
                List<ToolCall> calls = parseToolCalls(toolCallsJson);
                messages.add(ChatMessage.assistantWithToolCalls(content, calls));
            } else if ("assistant".equals(role)) {
                messages.add(ChatMessage.assistant(content == null ? "" : content));
            } else if ("user".equals(role)) {
                messages.add(ChatMessage.user(content));
            }
            // system 每轮重新生成，不从库里取
        }
        // 关键修剪：开头不能是 tool，否则网关 400
        while (!messages.isEmpty() && "tool".equals(messages.get(0).getRole())) {
            messages.remove(0);
        }
        return messages;
    }

    private List<ToolCall> parseToolCalls(String json) {
        try {
            ToolCall[] arr = mapper.readValue(json, ToolCall[].class);
            return new ArrayList<ToolCall>(Arrays.asList(arr));
        } catch (Exception ex) {
            log.warn("反序列化 tool_calls 失败: {}", json, ex);
            return new ArrayList<ToolCall>();
        }
    }

    /** 给前端展示的历史（去掉 tool 消息）。 */
    public List<Map<String, Object>> loadMessagesForUi(String sessionId) {
        return jdbc.queryForList(
                "SELECT role, content, tool_calls_json, created_at FROM agent_message "
                        + "WHERE session_id = ? AND role IN ('user','assistant') "
                        + "ORDER BY id ASC", sessionId);
    }

    // ---------------------------------------------------------- 挂起动作

    public void savePending(PendingAction pa) {
        // 同一会话只保留一个待处理动作，避免错串
        jdbc.update("UPDATE agent_pending_action SET status = 'ABANDONED' "
                + "WHERE session_id = ? AND status = 'WAITING'", pa.getSessionId());
        jdbc.update("INSERT INTO agent_pending_action "
                        + "(session_id, type, tool_call_id, tool_name, args_json) "
                        + "VALUES (?, ?, ?, ?, ?)",
                pa.getSessionId(), pa.getType(), pa.getToolCallId(),
                pa.getToolName(), pa.getArgsJson());
    }

    public PendingAction loadPending(String sessionId) {
        List<Map<String, Object>> rows = jdbc.queryForList(
                "SELECT id, session_id, type, tool_call_id, tool_name, args_json "
                        + "FROM agent_pending_action WHERE session_id = ? AND status = 'WAITING' "
                        + "ORDER BY id DESC LIMIT 1", sessionId);
        if (rows.isEmpty()) {
            return null;
        }
        Map<String, Object> row = rows.get(0);
        PendingAction pa = new PendingAction();
        pa.setId(((Number) row.get("id")).longValue());
        pa.setSessionId((String) row.get("session_id"));
        pa.setType((String) row.get("type"));
        pa.setToolCallId((String) row.get("tool_call_id"));
        pa.setToolName((String) row.get("tool_name"));
        pa.setArgsJson((String) row.get("args_json"));
        return pa;
    }

    public void clearPending(Long id) {
        jdbc.update("UPDATE agent_pending_action SET status = 'DONE' WHERE id = ?", id);
    }

    // ---------------------------------------------------------- 审计

    public void logToolCall(String sessionId, String userId, String toolName, String argsJson,
                            String resultText, boolean isError, long elapsedMs) {
        try {
            jdbc.update("INSERT INTO agent_tool_call "
                            + "(session_id, user_id, tool_name, args_json, result_text, is_error, elapsed_ms) "
                            + "VALUES (?, ?, ?, ?, ?, ?, ?)",
                    sessionId, userId, toolName, trim(argsJson, 60000),
                    trim(resultText, 60000), isError ? 1 : 0, elapsedMs);
        } catch (RuntimeException ex) {
            log.warn("审计写入失败（不影响主流程）", ex);
        }
    }

    private String trim(String s, int max) {
        if (s == null) {
            return null;
        }
        return s.length() <= max ? s : s.substring(0, max);
    }
}
```

## 4. `sse/SseHub.java` —— 事件推送

```java
package com.acme.simagent.sse;

import java.io.IOException;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;
import org.springframework.web.servlet.mvc.method.annotation.SseEmitter;

import com.fasterxml.jackson.databind.node.ObjectNode;

/**
 * 按 sessionId 维护 SSE 连接。
 * 事件类型：session / thinking / token / tool_call / tool_result /
 *          progress / plan_step / need_input / need_confirm / done / error / idle
 */
@Component
public class SseHub {

    private static final Logger log = LoggerFactory.getLogger(SseHub.class);
    private final Map<String, SseEmitter> emitters = new ConcurrentHashMap<String, SseEmitter>();

    /** 创建连接（0 = 不超时，由业务主动 complete）。 */
    public SseEmitter create(final String sessionId) {
        SseEmitter old = emitters.remove(sessionId);
        if (old != null) {
            old.complete();
        }
        final SseEmitter emitter = new SseEmitter(0L);
        emitter.onCompletion(new Runnable() {
            @Override
            public void run() { emitters.remove(sessionId); }
        });
        emitter.onTimeout(new Runnable() {
            @Override
            public void run() { emitters.remove(sessionId); }
        });
        emitter.onError(new java.util.function.Consumer<Throwable>() {
            @Override
            public void accept(Throwable t) { emitters.remove(sessionId); }
        });
        emitters.put(sessionId, emitter);
        return emitter;
    }

    public void emit(String sessionId, String event, ObjectNode payload) {
        SseEmitter emitter = emitters.get(sessionId);
        if (emitter == null) {
            log.debug("[SSE] 无连接，丢弃事件 session={} event={}", sessionId, event);
            return;
        }
        try {
            emitter.send(SseEmitter.event().name(event).data(payload.toString()));
        } catch (IOException | IllegalStateException ex) {
            log.debug("[SSE] 发送失败，移除连接 session={}", sessionId);
            emitters.remove(sessionId);
        }
    }

    public void complete(String sessionId) {
        SseEmitter emitter = emitters.remove(sessionId);
        if (emitter != null) {
            emitter.complete();
        }
    }

    public boolean isConnected(String sessionId) {
        return emitters.containsKey(sessionId);
    }
}
```

## 5. `simtask/SimTaskService.java` —— 任务落库（对接你现有业务）

```java
package com.acme.simagent.simtask;

import java.sql.PreparedStatement;
import java.sql.SQLException;
import java.sql.Statement;
import java.util.List;
import java.util.Map;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.core.PreparedStatementCreator;
import org.springframework.jdbc.support.GeneratedKeyHolder;
import org.springframework.jdbc.support.KeyHolder;
import org.springframework.stereotype.Service;

/** 仿真任务读写。如果你已有 SimTaskService，直接沿用并补上 mcp_job_id 字段即可。 */
@Service
public class SimTaskService {

    private static final Logger log = LoggerFactory.getLogger(SimTaskService.class);

    private final JdbcTemplate jdbc;

    public SimTaskService(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    public Long createTask(final String sessionId, final String userId, final String plugin,
                           final String workflowId, final String paramsJson) {
        final String sql = "INSERT INTO sim_task "
                + "(session_id, user_id, source, plugin, workflow_id, params_json, status) "
                + "VALUES (?, ?, 'AGENT', ?, ?, ?, 'CREATED')";
        KeyHolder keyHolder = new GeneratedKeyHolder();
        jdbc.update(new PreparedStatementCreator() {
            @Override
            public PreparedStatement createPreparedStatement(java.sql.Connection con)
                    throws SQLException {
                PreparedStatement ps = con.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS);
                ps.setString(1, sessionId);
                ps.setString(2, userId);
                ps.setString(3, plugin);
                ps.setString(4, workflowId);
                ps.setString(5, paramsJson);
                return ps;
            }
        }, keyHolder);
        Number key = keyHolder.getKey();
        return key == null ? null : key.longValue();
    }

    public void bindJob(Long taskId, String plugin, String jobId) {
        jdbc.update("UPDATE sim_task SET mcp_job_id = ?, status = 'QUEUED' WHERE id = ?",
                jobId, taskId);
    }

    public void markFailed(Long taskId, String error) {
        jdbc.update("UPDATE sim_task SET status = 'FAILED', error_message = ?, "
                        + "finished_at = NOW() WHERE id = ?",
                error == null ? "" : error.substring(0, Math.min(1000, error.length())), taskId);
    }

    public void updateStatus(Long taskId, String status, int progress, String currentStep,
                            String outputsJson, String error) {
        boolean terminal = "SUCCEEDED".equals(status) || "FAILED".equals(status)
                || "CANCELED".equals(status);
        jdbc.update("UPDATE sim_task SET status = ?, progress = ?, current_step = ?, "
                        + "outputs_json = COALESCE(?, outputs_json), "
                        + "error_message = COALESCE(?, error_message), "
                        + "finished_at = " + (terminal ? "NOW()" : "finished_at")
                        + " WHERE id = ?",
                status, progress, currentStep, outputsJson, error, taskId);
    }

    public void addStep(Long taskId, int index, String name, String status, String message) {
        jdbc.update("INSERT INTO sim_task_step (task_id, step_index, step_name, status, message) "
                + "VALUES (?, ?, ?, ?, ?)", taskId, index, name, status, message);
    }

    /** 轮询器用：所有未终止且已绑定 job 的任务。 */
    public List<Map<String, Object>> findUnfinished() {
        return jdbc.queryForList(
                "SELECT id, session_id, plugin, mcp_job_id, status FROM sim_task "
                        + "WHERE mcp_job_id IS NOT NULL "
                        + "AND status IN ('CREATED','QUEUED','RUNNING') "
                        + "AND created_at > DATE_SUB(NOW(), INTERVAL 1 DAY) "
                        + "ORDER BY id ASC LIMIT 50");
    }

    /** 内置工具 query_my_tasks 的返回文本（直接给大模型看）。 */
    public String listRecentAsText(String userId, int limit, String status) {
        int size = Math.max(1, Math.min(30, limit));
        List<Map<String, Object>> rows;
        if (status == null || status.isEmpty()) {
            rows = jdbc.queryForList(
                    "SELECT id, plugin, workflow_id, status, progress, params_json, "
                            + "outputs_json, created_at FROM sim_task "
                            + "WHERE user_id = ? ORDER BY id DESC LIMIT ?", userId, size);
        } else {
            rows = jdbc.queryForList(
                    "SELECT id, plugin, workflow_id, status, progress, params_json, "
                            + "outputs_json, created_at FROM sim_task "
                            + "WHERE user_id = ? AND status = ? ORDER BY id DESC LIMIT ?",
                    userId, status, size);
        }
        if (rows.isEmpty()) {
            return "该用户没有历史仿真任务。";
        }
        StringBuilder sb = new StringBuilder("历史任务（最新在前）：\n");
        for (Map<String, Object> r : rows) {
            sb.append("- 任务 #").append(r.get("id"))
              .append(" | ").append(r.get("plugin")).append('.').append(r.get("workflow_id"))
              .append(" | ").append(r.get("status"))
              .append(" ").append(r.get("progress")).append("%")
              .append(" | ").append(r.get("created_at")).append('\n')
              .append("  参数: ").append(cut(r.get("params_json"), 400)).append('\n');
            if (r.get("outputs_json") != null) {
                sb.append("  产物: ").append(cut(r.get("outputs_json"), 300)).append('\n');
            }
        }
        return sb.toString();
    }

    private String cut(Object o, int max) {
        if (o == null) {
            return "-";
        }
        String s = String.valueOf(o);
        return s.length() <= max ? s : s.substring(0, max) + "...";
    }
}
```

## 6. `simtask/SimTaskPoller.java` —— 异步任务回写

<aside>
⏱️

真实仿真可能要跑几十分钟，不可能卡在 HTTP 请求里。这个轮询器让任务状态**脘离对话生命周期**：用户关了网页，任务仍会被跟进并回写到业务库。

</aside>

```java
package com.acme.simagent.simtask;

import java.util.List;
import java.util.Map;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import com.acme.simagent.mcp.McpClient;
import com.acme.simagent.mcp.McpClientManager;
import com.acme.simagent.mcp.McpToolResult;
import com.acme.simagent.sse.SseHub;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;

/** 每 5 秒把 MCP 侧的作业状态同步到 sim_task，并推送给在线前端。 */
@Component
public class SimTaskPoller {

    private static final Logger log = LoggerFactory.getLogger(SimTaskPoller.class);

    private final SimTaskService simTaskService;
    private final McpClientManager mcpManager;
    private final SseHub sseHub;
    private final ObjectMapper mapper = new ObjectMapper();

    public SimTaskPoller(SimTaskService simTaskService, McpClientManager mcpManager,
                         SseHub sseHub) {
        this.simTaskService = simTaskService;
        this.mcpManager = mcpManager;
        this.sseHub = sseHub;
    }

    @Scheduled(fixedDelay = 5000L, initialDelay = 10000L)
    public void poll() {
        List<Map<String, Object>> tasks;
        try {
            tasks = simTaskService.findUnfinished();
        } catch (RuntimeException ex) {
            log.warn("[Poller] 查询待完成任务失败: {}", ex.getMessage());
            return;
        }
        for (Map<String, Object> task : tasks) {
            try {
                syncOne(task);
            } catch (RuntimeException ex) {
                log.warn("[Poller] 同步任务 {} 失败: {}", task.get("id"), ex.getMessage());
            }
        }
    }

    private void syncOne(Map<String, Object> task) {
        Long taskId = ((Number) task.get("id")).longValue();
        String plugin = (String) task.get("plugin");
        String jobId = (String) task.get("mcp_job_id");
        String sessionId = (String) task.get("session_id");

        McpClient client = mcpManager.get(plugin);
        if (client == null) {
            return;
        }
        ObjectNode args = mapper.createObjectNode();
        args.put("job_id", jobId);
        args.put("logs_tail_lines", 5);
        McpToolResult res = client.callTool("get_job_status", args, 20000L, null);
        JsonNode st = res.getStructured();
        if (st == null) {
            return;
        }

        String status = st.path("status").asText("RUNNING");
        int progress = (int) Math.round(st.path("progress").asDouble(0D));
        String currentStep = st.path("current_step").asText("");
        String outputs = st.hasNonNull("outputs") ? st.get("outputs").toString() : null;
        String error = st.hasNonNull("error_message") ? st.get("error_message").asText() : null;

        simTaskService.updateStatus(taskId, status, progress, currentStep, outputs, error);

        if (sessionId != null && sseHub.isConnected(sessionId)) {
            ObjectNode ev = mapper.createObjectNode();
            ev.put("taskId", taskId);
            ev.put("jobId", jobId);
            ev.put("status", status);
            ev.put("progress", progress);
            ev.put("message", currentStep);
            sseHub.emit(sessionId, "progress", ev);
        }
    }
}
```

```java
    /** 发送一条消息，返回 SSE 事件流。 */
    @PostMapping(value = "/chat", produces = "text/event-stream;charset=UTF-8")
    public SseEmitter chat(@RequestBody final Map<String, String> body,
                           @RequestHeader(value = "X-User-Id", required = false) String header) {
        final String userId = header != null ? header
                : (body.get("userId") != null ? body.get("userId") : "anonymous");
        final String message = body.get("message");
        final String sessionId = store.ensureSession(body.get("sessionId"), userId, message);

        SseEmitter emitter = sseHub.create(sessionId);
        ObjectNode init = mapper.createObjectNode();
        init.put("sessionId", sessionId);
        sseHub.emit(sessionId, "session", init);

        agentExecutor.execute(new Runnable() {
            @Override
            public void run() {
                try {
                    orchestrator.chat(sessionId, userId, message);
                } catch (RuntimeException ex) {
                    log.error("[API] chat 异常", ex);
                } finally {
                    sseHub.complete(sessionId);
                }
            }
        });
        return emitter;
    }

    /** 用户填完补参表单，或点了确认/拒绝。 */
    @PostMapping(value = "/resume", produces = "text/event-stream;charset=UTF-8")
    public SseEmitter resume(@RequestBody final Map<String, Object> body,
                            @RequestHeader(value = "X-User-Id", required = false) String header) {
        final String userId = header != null ? header : "anonymous";
        final String sessionId = String.valueOf(body.get("sessionId"));
        final JsonNode payload = mapper.valueToTree(
                body.get("payload") == null ? new HashMap<String, Object>() : body.get("payload"));

        SseEmitter emitter = sseHub.create(sessionId);
        agentExecutor.execute(new Runnable() {
            @Override
            public void run() {
                try {
                    orchestrator.resume(sessionId, userId, payload);
                } catch (RuntimeException ex) {
                    log.error("[API] resume 异常", ex);
                } finally {
                    sseHub.complete(sessionId);
                }
            }
        });
        return emitter;
    }

    /** 历史消息（刷新页面后恢复对话）。 */
    @GetMapping(value = "/sessions/{sessionId}/messages",
            produces = MediaType.APPLICATION_JSON_VALUE)

        out.put("toolCount", mcpManager.allTools().size());
        out.put("capabilityGraph", capabilities.getGraphText());
        return out;
    }
```

```html
// EventSource 不支持 POST，所以用 fetch 读流并自己拆 SSE 帧
async function stream(url, body) {
  const resp = await fetch(url, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', 'X-User-Id': USER_ID },
    body: JSON.stringify(body)
  });
  const reader = resp.body.getReader();
  const decoder = new TextDecoder('utf-8');
  let buffer = '';
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    let idx;
    while ((idx = buffer.indexOf('\n\n')) >= 0) {
      const frame = buffer.slice(0, idx);
      buffer = buffer.slice(idx + 2);
      let event = 'message', data = '';
      frame.split('\n').forEach(line => {
        if (line.startsWith('event:')) event = line.slice(6).trim();
        else if (line.startsWith('data:')) data += line.slice(5).trim();
      });
      if (data) handle(event, JSON.parse(data));
    }
  }
}
```

# 03 · Java 代码（一）手写 MCP 客户端 + LLM 客户端

<aside>
🎯

这一页是整个方案技术含量最高的部分：**不用任何框架，用 OkHttp + Jackson 手写一个合规的 MCP Streamable HTTP 客户端**（含握手、会话头、SSE 双模式响应、进度通知、会话过期重连），以及一个支持 `tool_calls` 分片聚合的流式 LLM 客户端。Java 8 兼容，无 text block、无 `var`、无 `java.net.http`。

</aside>

## 4. `mcp/StreamableHttpMcpTransport.java` —— 核心！

<aside>
📚

实现严格对着 MCP `2025-06-18` 传输规范写的，重点：① POST 必须带 `Accept: application/json, text/event-stream`；② `initialize` 响应头里的 `Mcp-Session-Id` 之后每个请求都要带；③ 必须带 `MCP-Protocol-Version`；④ 响应可能是单个 JSON 也可能是 SSE 流，两种都要处理；⑤ 404/400 代表会话失效，需重新 initialize。

</aside>

```java
package com.acme.simagent.mcp;

import java.io.IOException;
import java.util.Collections;
import java.util.Map;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicLong;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;

import okhttp3.MediaType;
import okhttp3.OkHttpClient;
import okhttp3.Request;
import okhttp3.RequestBody;
import okhttp3.Response;
import okio.BufferedSource;

/** 手写 MCP Streamable HTTP 客户端（协议版本 2025-06-18）。 */
public class StreamableHttpMcpTransport implements McpTransport {

    private static final Logger log = LoggerFactory.getLogger(StreamableHttpMcpTransport.class);
    private static final MediaType JSON_MEDIA = MediaType.parse("application/json; charset=utf-8");
    public static final String PROTOCOL_VERSION = "2025-06-18";

    private final String name;
    private final String url;
    private final Map<String, String> extraHeaders;
    private final ObjectMapper mapper = new ObjectMapper();
    private final OkHttpClient baseClient;
    private final AtomicLong idGen = new AtomicLong(0L);
    private final Object connectLock = new Object();

    private volatile String sessionId;
    private volatile boolean initialized;
    private volatile String serverName = "";
    private volatile String serverInstructions = "";

    public StreamableHttpMcpTransport(String name, String url, Map<String, String> headers) {
        this.name = name;
        this.url = url;
        this.extraHeaders = headers == null
                ? Collections.<String, String>emptyMap() : headers;
        this.baseClient = new OkHttpClient.Builder()
                .connectTimeout(10, TimeUnit.SECONDS)
                .writeTimeout(30, TimeUnit.SECONDS)
                .readTimeout(300, TimeUnit.SECONDS)   // SSE 长连接，单次调用会再覆盖
                .retryOnConnectionFailure(true)
                .build();
    }

    // ---------------------------------------------------------------- 握手

    @Override
    public void connect() throws Exception {
        synchronized (connectLock) {
            sessionId = null;
            initialized = false;

            ObjectNode params = mapper.createObjectNode();
            params.put("protocolVersion", PROTOCOL_VERSION);
            ObjectNode caps = params.putObject("capabilities");
            caps.putObject("roots").put("listChanged", false);
            // 注意：我们不声明 sampling / elicitation，表示不接受服务端反向调用，
            // 让所有交互都回到我们自己可控的 ask_user 机制上。
            ObjectNode clientInfo = params.putObject("clientInfo");
            clientInfo.put("name", "sim-agent-java");
            clientInfo.put("version", "1.0.0");

            RpcResponse res = post("initialize", params, 20000L, null);
            if (res.sessionId != null && res.sessionId.length() > 0) {
                sessionId = res.sessionId;
            }
            serverName = res.result.path("serverInfo").path("name").asText(name);
            serverInstructions = res.result.path("instructions").asText("");
            initialized = true;

            // 规范要求：握手完成后必须发送 initialized 通知
            notifyServer("notifications/initialized", mapper.createObjectNode());
            log.info("[MCP] connected server={} impl={} session={} protocol={}",
                    name, serverName, sessionId, PROTOCOL_VERSION);
        }
    }

    // ---------------------------------------------------------------- 请求

    @Override
    public JsonNode request(String method, ObjectNode params, long timeoutMs,
                           NotificationHandler handler) throws Exception {
        if (!initialized) {
            connect();
        }
        try {
            return post(method, params, timeoutMs, handler).result;
        } catch (McpSessionExpiredException expired) {
            log.warn("[MCP] session expired server={}, reconnect and retry once: {}",
                    name, expired.getMessage());
            connect();
            return post(method, params, timeoutMs, handler).result;
        }
    }

    @Override
    public void notifyServer(String method, ObjectNode params) throws Exception {
        ObjectNode body = mapper.createObjectNode();
        body.put("jsonrpc", "2.0");
        body.put("method", method);
        if (params != null) {
            body.set("params", params);
        }
        Response resp = newCall(body, 15000L).execute();
        try {
            // 规范：只发 notification 时，服务端必须返回 202 Accepted 且无 body
            if (resp.code() != 202 && !resp.isSuccessful()) {
                log.warn("[MCP] notification {} got HTTP {}", method, resp.code());
            }
        } finally {
            resp.close();
        }
    }

    private static final class RpcResponse {
        JsonNode result;
        String sessionId;
    }

    private RpcResponse post(String method, ObjectNode params, long timeoutMs,
                             NotificationHandler handler) throws Exception {
        long id = idGen.incrementAndGet();
        ObjectNode body = mapper.createObjectNode();
        body.put("jsonrpc", "2.0");
        body.put("id", id);
        body.put("method", method);
        if (params != null) {
            body.set("params", params);
        }

        Response resp = newCall(body, timeoutMs).execute();
        try {
            int code = resp.code();
            RpcResponse out = new RpcResponse();
            out.sessionId = resp.header("Mcp-Session-Id");

            if (code == 404 || code == 400) {
                throw new McpSessionExpiredException("HTTP " + code + " body=" + safeBody(resp));
            }
            if (code == 401 || code == 403) {
                throw new McpException("MCP 鉴权失败 HTTP " + code
                        + "，请检查 agent.mcp[].headers 里的 token", code);
            }
            if (!resp.isSuccessful()) {
                throw new McpException("MCP HTTP " + code + " body=" + safeBody(resp), code);
            }

            String contentType = resp.header("Content-Type", "");
            if (contentType != null && contentType.startsWith("text/event-stream")) {
                out.result = readSseUntilResponse(resp, id, handler);
            } else if (contentType != null && contentType.startsWith("application/json")) {
                JsonNode node = mapper.readTree(resp.body().string());
                out.result = extractResult(node);
            } else if (code == 202) {
                out.result = mapper.createObjectNode();
            } else {
                throw new McpException("MCP 返回了意外的 Content-Type: " + contentType);
            }
            return out;
        } finally {
            resp.close();
        }
    }

    private okhttp3.Call newCall(ObjectNode body, long timeoutMs) throws IOException {
        Request.Builder rb = new Request.Builder()
                .url(url)
                .post(RequestBody.create(JSON_MEDIA, mapper.writeValueAsString(body)))
                // 规范强制：两种响应都要能接
                .header("Accept", "application/json, text/event-stream")
                .header("MCP-Protocol-Version", PROTOCOL_VERSION);
        if (sessionId != null) {
            rb.header("Mcp-Session-Id", sessionId);
        }
        for (Map.Entry<String, String> e : extraHeaders.entrySet()) {
            rb.header(e.getKey(), e.getValue());
        }
        OkHttpClient client = baseClient.newBuilder()
                .readTimeout(timeoutMs, TimeUnit.MILLISECONDS)
                .callTimeout(timeoutMs + 10000L, TimeUnit.MILLISECONDS)
                .build();
        return client.newCall(rb.build());
    }

    /** 读 SSE 流，直到拿到与本次 id 匹配的 response；途中的通知交给 handler。 */
    private JsonNode readSseUntilResponse(Response resp, long expectedId,
                                          NotificationHandler handler) throws Exception {
        BufferedSource source = resp.body().source();
        StringBuilder data = new StringBuilder();
        while (true) {
            String line = source.readUtf8Line();
            if (line == null) {
                throw new McpException("SSE 流提前结束，未收到 id=" + expectedId + " 的响应");
            }
            if (line.startsWith(":")) {
                continue;                       // 注释行 / keep-alive
            }
            if (line.startsWith("data:")) {
                data.append(line.substring(5).trim());
                continue;
            }
            if (line.length() > 0) {
                continue;                       // event: / id: / retry: 等字段忽略
            }
            // 空行 = 一个完整事件结束
            if (data.length() == 0) {
                continue;
            }
            String payload = data.toString();
            data.setLength(0);
            JsonNode node = mapper.readTree(payload);

            if (node.hasNonNull("method") && !node.has("id")) {
                // 服务端通知（进度/日志）
                if (handler != null) {
                    try {
                        handler.onNotification(node.get("method").asText(), node.path("params"));
                    } catch (RuntimeException ex) {
                        log.warn("[MCP] notification handler error", ex);
                    }
                }
                continue;
            }
            if (node.hasNonNull("method") && node.has("id")) {
                // 服务端反向请求（sampling/elicitation）：我们不支持，礼貌拒绝
                replyMethodNotFound(node.get("id"), node.get("method").asText());
                continue;
            }
            if (node.has("id") && node.get("id").asLong(-1L) == expectedId) {
                return extractResult(node);
            }
            log.debug("[MCP] 忽略无关消息: {}", payload);
        }
    }

    private void replyMethodNotFound(JsonNode id, String method) {
        try {
            ObjectNode body = mapper.createObjectNode();
            body.put("jsonrpc", "2.0");
            body.set("id", id);
            ObjectNode err = body.putObject("error");
            err.put("code", -32601);
            err.put("message", "client does not support " + method);
            Response r = newCall(body, 10000L).execute();
            r.close();
        } catch (Exception ex) {
            log.warn("[MCP] 回复服务端请求失败 method={}", method, ex);
        }
    }

    private JsonNode extractResult(JsonNode node) {
        if (node.has("error")) {
            JsonNode err = node.get("error");
            throw new McpException("MCP 服务端错误: " + err.path("message").asText()
                    + " data=" + err.path("data").toString(), err.path("code").asInt(0));
        }
        return node.has("result") ? node.get("result") : mapper.createObjectNode();
    }

    private String safeBody(Response resp) {
        try {
            return resp.body() == null ? "" : resp.body().string();
        } catch (Exception ex) {
            return "<unreadable>";
        }
    }

    // ---------------------------------------------------------------- 其他

    @Override
    public boolean isAlive() { return initialized; }

    @Override
    public String getServerName() { return serverName; }

    @Override
    public String getServerInstructions() { return serverInstructions; }

    @Override
    public void close() {
        if (sessionId == null) {
            return;
        }
        try {
            Request.Builder rb = new Request.Builder().url(url).delete()
                    .header("Mcp-Session-Id", sessionId)
                    .header("MCP-Protocol-Version", PROTOCOL_VERSION);
            for (Map.Entry<String, String> e : extraHeaders.entrySet()) {
                rb.header(e.getKey(), e.getValue());
            }
            Response r = baseClient.newCall(rb.build()).execute();
            r.close();
        } catch (Exception ex) {
            log.debug("[MCP] 关闭会话失败（可忽略）", ex);
        } finally {
            initialized = false;
            sessionId = null;
        }
    }
}
```

## 5. `mcp/StdioMcpTransport.java` —— 子进程模式（本地调试首选）

```java
package com.acme.simagent.mcp;

import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.File;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicLong;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;

/** stdio 传输：把 MCP Server 当子进程拉起，按行交换 JSON-RPC。 */
public class StdioMcpTransport implements McpTransport {

    private static final Logger log = LoggerFactory.getLogger(StdioMcpTransport.class);

    private final String name;
    private final List<String> command;
    private final File workDir;
    private final ObjectMapper mapper = new ObjectMapper();
    private final AtomicLong idGen = new AtomicLong(0L);
    private final Map<Long, CompletableFuture<JsonNode>> pending =
            new ConcurrentHashMap<Long, CompletableFuture<JsonNode>>();
    private final Map<Long, NotificationHandler> handlers =
            new ConcurrentHashMap<Long, NotificationHandler>();

    private Process process;
    private BufferedWriter writer;
    private volatile boolean running;
    private volatile String serverName = "";
    private volatile String serverInstructions = "";

    public StdioMcpTransport(String name, String command, List<String> args, String workDir) {
        this.name = name;
        List<String> cmd = new ArrayList<String>();
        cmd.add(command);
        if (args != null) {
            cmd.addAll(args);
        }
        this.command = cmd;
        this.workDir = (workDir == null || workDir.isEmpty()) ? null : new File(workDir);
    }

    @Override
    public synchronized void connect() throws Exception {
        ProcessBuilder pb = new ProcessBuilder(command);
        if (workDir != null) {
            pb.directory(workDir);
        }
        pb.redirectErrorStream(false);
        process = pb.start();
        writer = new BufferedWriter(new OutputStreamWriter(
                process.getOutputStream(), StandardCharsets.UTF_8));
        running = true;

        startReader(new BufferedReader(new InputStreamReader(
                process.getInputStream(), StandardCharsets.UTF_8)));
        startStderrDrain(new BufferedReader(new InputStreamReader(
                process.getErrorStream(), StandardCharsets.UTF_8)));

        ObjectNode params = mapper.createObjectNode();
        params.put("protocolVersion", StreamableHttpMcpTransport.PROTOCOL_VERSION);
        params.putObject("capabilities");
        ObjectNode info = params.putObject("clientInfo");
        info.put("name", "sim-agent-java");
        info.put("version", "1.0.0");

        JsonNode result = request("initialize", params, 20000L, null);
        serverName = result.path("serverInfo").path("name").asText(name);
        serverInstructions = result.path("instructions").asText("");
        notifyServer("notifications/initialized", mapper.createObjectNode());
        log.info("[MCP-stdio] connected {} impl={}", name, serverName);
    }

    private void startReader(final BufferedReader reader) {
        Thread t = new Thread(new Runnable() {
            @Override
            public void run() {
                try {
                    String line;
                    while (running && (line = reader.readLine()) != null) {
                        if (line.trim().isEmpty()) {
                            continue;
                        }
                        dispatch(mapper.readTree(line));
                    }
                } catch (Exception ex) {
                    if (running) {
                        log.warn("[MCP-stdio] reader stopped: {}", ex.getMessage());
                    }
                } finally {
                    running = false;
                    for (CompletableFuture<JsonNode> f : pending.values()) {
                        f.completeExceptionally(new McpException("stdio 进程已退出"));
                    }
                    pending.clear();
                }
            }
        }, "mcp-stdio-reader-" + name);
        t.setDaemon(true);
        t.start();
    }

    private void startStderrDrain(final BufferedReader reader) {
        Thread t = new Thread(new Runnable() {
            @Override
            public void run() {
                try {
                    String line;
                    while (running && (line = reader.readLine()) != null) {
                        log.debug("[MCP-stdio:{}] {}", name, line);
                    }
                } catch (Exception ignored) {
                    // 子进程退出即可
                }
            }
        }, "mcp-stdio-stderr-" + name);
        t.setDaemon(true);
        t.start();
    }

    private void dispatch(JsonNode node) {
        if (node.has("id") && !node.hasNonNull("method")) {
            long id = node.get("id").asLong(-1L);
            CompletableFuture<JsonNode> future = pending.remove(id);
            handlers.remove(id);
            if (future == null) {
                return;
            }
            if (node.has("error")) {
                JsonNode err = node.get("error");
                future.completeExceptionally(new McpException(
                        err.path("message").asText(), err.path("code").asInt(0)));
            } else {
                future.complete(node.path("result"));
            }
            return;
        }
        if (node.hasNonNull("method")) {
            String method = node.get("method").asText();
            long token = node.path("params").path("progressToken").asLong(-1L);
            NotificationHandler h = token > 0 ? handlers.get(token) : null;
            if (h == null) {
                for (NotificationHandler each : handlers.values()) {
                    each.onNotification(method, node.path("params"));
                }
            } else {
                h.onNotification(method, node.path("params"));
            }
        }
    }

    @Override
    public JsonNode request(String method, ObjectNode params, long timeoutMs,
                           NotificationHandler handler) throws Exception {
        long id = idGen.incrementAndGet();
        ObjectNode body = mapper.createObjectNode();
        body.put("jsonrpc", "2.0");
        body.put("id", id);
        body.put("method", method);
        if (params != null) {
            body.set("params", params);
        }
        CompletableFuture<JsonNode> future = new CompletableFuture<JsonNode>();
        pending.put(id, future);
        if (handler != null) {
            handlers.put(id, handler);
        }
        writeLine(mapper.writeValueAsString(body));
        try {
            return future.get(timeoutMs, TimeUnit.MILLISECONDS);
        } finally {
            pending.remove(id);
            handlers.remove(id);
        }
    }

    @Override
    public void notifyServer(String method, ObjectNode params) throws Exception {
        ObjectNode body = mapper.createObjectNode();
        body.put("jsonrpc", "2.0");
        body.put("method", method);
        if (params != null) {
            body.set("params", params);
        }
        writeLine(mapper.writeValueAsString(body));
    }

    private synchronized void writeLine(String json) throws Exception {
        if (!running || writer == null) {
            throw new McpException("stdio 传输未就绪");
        }
        writer.write(json);
        writer.write("\n");
        writer.flush();
    }

    @Override
    public boolean isAlive() { return running && process != null && process.isAlive(); }

    @Override
    public String getServerName() { return serverName; }

    @Override
    public String getServerInstructions() { return serverInstructions; }

    @Override
    public void close() {
        running = false;
        if (process != null) {
            process.destroy();
        }
    }
}
```

```java
package com.acme.simagent.mcp;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.UUID;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;

/** 单个 MCP Server 的客户端：工具发现 + 调用 + 资源读取。 */
public class McpClient {

    private static final Logger log = LoggerFactory.getLogger(McpClient.class);

    /** 仿真进度监听（最终会转成 SSE 推给前端）。 */
    public interface ProgressListener {
        void onProgress(double progress, double total, String message);
    }

    private final String serverName;
    private final McpTransport transport;
    private final long defaultTimeoutMs;
    private final ObjectMapper mapper = new ObjectMapper();
    private volatile List<McpTool> tools = Collections.emptyList();

    public McpClient(String serverName, McpTransport transport, long defaultTimeoutMs) {
        this.serverName = serverName;
        this.transport = transport;
        this.defaultTimeoutMs = defaultTimeoutMs;
    }

    public String getServerName() { return serverName; }
    public List<McpTool> getTools() { return tools; }
    public String getInstructions() { return transport.getServerInstructions(); }
    public boolean isAlive() { return transport.isAlive(); }

    public void connect() throws Exception {
        transport.connect();
        refreshTools();
    }

    /** tools/list（自动翻页）。 */
    public void refreshTools() throws Exception {
        List<McpTool> collected = new ArrayList<McpTool>();
        String cursor = null;
        do {
            ObjectNode params = mapper.createObjectNode();
            if (cursor != null) {
                params.put("cursor", cursor);
            }
            JsonNode result = transport.request("tools/list", params, defaultTimeoutMs, null);
            for (JsonNode t : result.path("tools")) {
                boolean readOnly = t.path("annotations").path("readOnlyHint").asBoolean(false);
                collected.add(new McpTool(serverName,
                        t.path("name").asText(),
                        t.path("description").asText(""),
                        t.get("inputSchema"),
                        readOnly));
            }
            cursor = result.hasNonNull("nextCursor") ? result.get("nextCursor").asText() : null;
        } while (cursor != null && !cursor.isEmpty());
        this.tools = Collections.unmodifiableList(collected);
        log.info("[MCP] server={} tools={}", serverName, collected.size());
    }

    /** tools/call：带进度 token，把服务端进度通知回传给调用方。 */
    public McpToolResult callTool(String toolName, JsonNode arguments, long timeoutMs,
                                  final ProgressListener listener) {
        final String progressToken = "pt-" + UUID.randomUUID().toString();
        ObjectNode params = mapper.createObjectNode();
        params.put("name", toolName);
        params.set("arguments", arguments == null ? mapper.createObjectNode() : arguments);
        params.putObject("_meta").put("progressToken", progressToken);

        McpTransport.NotificationHandler handler = new McpTransport.NotificationHandler() {
            @Override
            public void onNotification(String method, JsonNode p) {
                if ("notifications/progress".equals(method) && listener != null
                        && progressToken.equals(p.path("progressToken").asText())) {
                    listener.onProgress(p.path("progress").asDouble(0D),
                            p.path("total").asDouble(100D),
                            p.path("message").asText(""));
                } else if ("notifications/message".equals(method)) {
                    log.debug("[MCP:{}] log {}", serverName, p.path("data").toString());
                }
            }
        };

        try {
            JsonNode result = transport.request("tools/call", params,
                    timeoutMs > 0 ? timeoutMs : defaultTimeoutMs, handler);
            StringBuilder text = new StringBuilder();
            for (JsonNode c : result.path("content")) {
                String type = c.path("type").asText();
                if ("text".equals(type)) {
                    text.append(c.path("text").asText()).append('\n');
                } else if ("resource_link".equals(type)) {
                    text.append("[资源] ").append(c.path("uri").asText()).append('\n');
                } else {
                    text.append('[').append(type).append(" 内容已省略]").append('\n');
                }
            }
            boolean isError = result.path("isError").asBoolean(false);
            JsonNode structured = result.get("structuredContent");
            if (text.length() == 0 && structured != null) {
                text.append(structured.toString());
            }
            return new McpToolResult(isError, text.toString().trim(), structured);
        } catch (McpException ex) {
            // 工具层错误不抛异常，而是变成结果回给大模型，让它自我修正
            return new McpToolResult(true, "工具调用失败: " + ex.getMessage(), null);
        } catch (Exception ex) {
            log.error("[MCP] callTool 异常 server={} tool={}", serverName, toolName, ex);
            return new McpToolResult(true,
                    "工具调用异常: " + ex.getClass().getSimpleName() + " " + ex.getMessage(), null);
        }
    }

    /** resources/read：读 SOP 等文本资源。 */
    public String readResource(String uri) throws Exception {
        ObjectNode params = mapper.createObjectNode();
        params.put("uri", uri);
        JsonNode result = transport.request("resources/read", params, defaultTimeoutMs, null);
        StringBuilder sb = new StringBuilder();
        for (JsonNode c : result.path("contents")) {
            sb.append(c.path("text").asText("")).append('\n');
        }
        return sb.toString();
    }

    public void close() {
        try {
            transport.close();
        } catch (Exception ignored) {
            // 关闭失败不影响主流程
        }
    }
}
```

## 7. `mcp/McpClientManager.java` —— 多插件生命周期 + 健康检查

```java
package com.acme.simagent.mcp;

import java.util.ArrayList;
import java.util.Collection;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

import javax.annotation.PostConstruct;
import javax.annotation.PreDestroy;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import com.acme.simagent.config.AgentProperties;

/** 管理所有仿真插件连接：异步预热、定时健康检查、工具刷新。 */
@Component
public class McpClientManager {

    private static final Logger log = LoggerFactory.getLogger(McpClientManager.class);

    private final AgentProperties props;
    private final Map<String, McpClient> clients = new ConcurrentHashMap<String, McpClient>();
    private final Map<String, String> lastErrors = new ConcurrentHashMap<String, String>();
    private final ExecutorService bootPool = Executors.newCachedThreadPool();

    public McpClientManager(AgentProperties props) {
        this.props = props;
    }

    @PostConstruct
    public void init() {
        for (AgentProperties.McpServer cfg : props.getMcp()) {
            if (!cfg.isEnabled()) {
                continue;
            }
            // 异步预热：即使某个仿真插件没启，也不能阻塞 Spring 启动
            bootPool.submit(new Runnable() {
                @Override
                public void run() {
                    connectWithRetry(cfg, 5);
                }
            });
        }
    }

    private void connectWithRetry(AgentProperties.McpServer cfg, int attempts) {
        for (int i = 1; i <= attempts; i++) {
            try {
                McpTransport transport;
                if ("stdio".equalsIgnoreCase(cfg.getTransport())) {
                    transport = new StdioMcpTransport(cfg.getName(), cfg.getCommand(),
                            cfg.getArgs(), cfg.getWorkDir());
                } else {
                    transport = new StreamableHttpMcpTransport(cfg.getName(), cfg.getUrl(),
                            cfg.getHeaders());
                }
                McpClient client = new McpClient(cfg.getName(), transport, cfg.getTimeoutMs());
                client.connect();
                clients.put(cfg.getName(), client);
                lastErrors.remove(cfg.getName());
                log.info("[MCP] 插件 {} 就绪，工具数={}", cfg.getName(), client.getTools().size());
                return;
            } catch (Exception ex) {
                lastErrors.put(cfg.getName(), ex.getMessage());
                log.warn("[MCP] 插件 {} 第 {} 次连接失败: {}", cfg.getName(), i, ex.getMessage());
                try {
                    Thread.sleep(Math.min(30000L, 2000L * i));   // 指数退避
                } catch (InterruptedException ie) {
                    Thread.currentThread().interrupt();
                    return;
                }
            }
        }
    }

    /** 每 60 秒：掉线的重连，在线的刷新工具列表。 */
    @Scheduled(fixedDelay = 60000L, initialDelay = 60000L)
    public void healthCheck() {
        for (AgentProperties.McpServer cfg : props.getMcp()) {
            if (!cfg.isEnabled()) {
                continue;
            }
            McpClient client = clients.get(cfg.getName());
            if (client == null || !client.isAlive()) {
                log.info("[MCP] 插件 {} 不可用，尝试重连", cfg.getName());
                connectWithRetry(cfg, 1);
                continue;
            }
            try {
                client.refreshTools();
            } catch (Exception ex) {
                lastErrors.put(cfg.getName(), ex.getMessage());
                log.warn("[MCP] 刷新工具失败 {}: {}", cfg.getName(), ex.getMessage());
            }
        }
    }

    public McpClient get(String serverName) {
        return clients.get(serverName);
    }

    public Collection<McpClient> all() {
        return clients.values();
    }

    public List<McpTool> allTools() {
        List<McpTool> list = new ArrayList<McpTool>();
        for (McpClient c : clients.values()) {
            list.addAll(c.getTools());
        }
        return list;
    }

    /** 给 /api/agent/health/mcp 用。 */
    public Map<String, Object> healthSnapshot() {
        Map<String, Object> out = new LinkedHashMap<String, Object>();
        for (AgentProperties.McpServer cfg : props.getMcp()) {
            Map<String, Object> item = new LinkedHashMap<String, Object>();
            McpClient c = clients.get(cfg.getName());
            item.put("enabled", cfg.isEnabled());
            item.put("transport", cfg.getTransport());
            item.put("url", cfg.getUrl());
            item.put("connected", c != null && c.isAlive());
            item.put("toolCount", c == null ? 0 : c.getTools().size());
            item.put("lastError", lastErrors.get(cfg.getName()));
            out.put(cfg.getName(), item);
        }
        return out;
    }

    @PreDestroy
    public void shutdown() {
        for (McpClient c : clients.values()) {
            c.close();
        }
        bootPool.shutdownNow();
    }
}
```

## 9. `llm/OpenAiCompatibleLlmClient.java` —— 流式 + `tool_calls` 分片聚合

<aside>
⚠️

流式下 `tool_calls` 是**按字符分片下发**的：`arguments` 会分成 `{"beam_`、`length_um":`、`200}` 这样的碎片，必须按 `index` 拼回去。这是手写 Agent 最容易翼的地方。

</aside>

```java
package com.acme.simagent.llm;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;
import java.util.concurrent.TimeUnit;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;

import com.acme.simagent.config.AgentProperties;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ArrayNode;
import com.fasterxml.jackson.databind.node.ObjectNode;

import okhttp3.MediaType;
import okhttp3.OkHttpClient;
import okhttp3.Request;
import okhttp3.RequestBody;
import okhttp3.Response;
import okio.BufferedSource;

/** OpenAI 兼容的大模型客户端（不依赖 Spring AI）。 */
@Component
public class OpenAiCompatibleLlmClient {

    private static final Logger log = LoggerFactory.getLogger(OpenAiCompatibleLlmClient.class);
    private static final MediaType JSON_MEDIA = MediaType.parse("application/json; charset=utf-8");

    /** 流式回调：只用于把正文 token 实时推给前端。 */
    public interface StreamHandler {
        void onToken(String token);
    }

    private final AgentProperties props;
    private final ObjectMapper mapper = new ObjectMapper();
    private final OkHttpClient http;

    public OpenAiCompatibleLlmClient(AgentProperties props) {
        this.props = props;
        this.http = new OkHttpClient.Builder()
                .connectTimeout(15, TimeUnit.SECONDS)
                .readTimeout(props.getLlm().getTimeoutMs(), TimeUnit.MILLISECONDS)
                .build();
    }

    // ------------------------------------------------------------ 对外接口

    public ChatMessage chat(List<ChatMessage> messages, List<ObjectNode> tools) throws Exception {
        return call(messages, tools, null);
    }

    public ChatMessage streamChat(List<ChatMessage> messages, List<ObjectNode> tools,
                                 StreamHandler handler) throws Exception {
        return call(messages, tools, handler);
    }

    // ------------------------------------------------------------ 内部实现

    private ChatMessage call(List<ChatMessage> messages, List<ObjectNode> tools,
                             StreamHandler handler) throws Exception {
        boolean stream = handler != null;
        ObjectNode body = mapper.createObjectNode();
        body.put("model", props.getLlm().getModel());
        body.put("temperature", props.getLlm().getTemperature());
        body.put("max_tokens", props.getLlm().getMaxTokens());
        body.put("stream", stream);
        body.set("messages", mapper.valueToTree(messages));
        if (tools != null && !tools.isEmpty()) {
            ArrayNode arr = body.putArray("tools");
            for (ObjectNode t : tools) {
                arr.add(t);
            }
            body.put("tool_choice", "auto");
            // 关键：禁止并行工具调用，一步一个，便于审批、幂等与错误定位
            body.put("parallel_tool_calls", false);
        }

        String url = props.getLlm().getBaseUrl().replaceAll("/+$", "") + "/chat/completions";
        Exception last = null;
        for (int attempt = 0; attempt <= props.getLlm().getMaxRetries(); attempt++) {
            try {
                Request req = new Request.Builder().url(url)
                        .post(RequestBody.create(JSON_MEDIA, mapper.writeValueAsString(body)))
                        .header("Authorization", "Bearer " + props.getLlm().getApiKey())
                        .header("Content-Type", "application/json")
                        .header("Accept", stream ? "text/event-stream" : "application/json")
                        .build();
                Response resp = http.newCall(req).execute();
                try {
                    if (resp.code() == 429 || resp.code() >= 500) {
                        throw new RuntimeException("LLM HTTP " + resp.code() + ": " + bodyText(resp));
                    }
                    if (!resp.isSuccessful()) {
                        // 4xx 不重试（参数/消息格式错误，重试也无用）
                        throw new IllegalStateException("LLM HTTP " + resp.code() + ": " + bodyText(resp));
                    }
                    return stream ? parseStream(resp, handler) : parseOnce(resp);
                } finally {
                    resp.close();
                }
            } catch (IllegalStateException fatal) {
                throw fatal;
            } catch (Exception ex) {
                last = ex;
                long backoff = (long) (Math.pow(2, attempt) * 800L);
                log.warn("LLM 调用失败（第 {} 次），{}ms 后重试: {}", attempt + 1, backoff, ex.getMessage());
                Thread.sleep(backoff);
            }
        }
        throw new IllegalStateException("LLM 调用多次失败", last);
    }

    private ChatMessage parseOnce(Response resp) throws Exception {
        JsonNode root = mapper.readTree(resp.body().string());
        JsonNode msg = root.path("choices").path(0).path("message");
        List<ToolCall> calls = new ArrayList<ToolCall>();
        for (JsonNode tc : msg.path("tool_calls")) {
            calls.add(new ToolCall(tc.path("id").asText(),
                    tc.path("function").path("name").asText(),
                    tc.path("function").path("arguments").asText("{}")));
        }
        String content = msg.hasNonNull("content") ? msg.get("content").asText() : "";
        return calls.isEmpty() ? ChatMessage.assistant(content)
                : ChatMessage.assistantWithToolCalls(content.isEmpty() ? null : content, calls);
    }

    /** 流式解析：累加 content，并按 index 拼回 tool_calls 碎片。 */
    private ChatMessage parseStream(Response resp, StreamHandler handler) throws Exception {
        BufferedSource source = resp.body().source();
        StringBuilder content = new StringBuilder();
        Map<Integer, ToolCallBuilder> builders = new TreeMap<Integer, ToolCallBuilder>();

        while (true) {
            String line = source.readUtf8Line();
            if (line == null) {
                break;
            }
            if (!line.startsWith("data:")) {
                continue;
            }
            String payload = line.substring(5).trim();
            if (payload.isEmpty()) {
                continue;
            }
            if ("[DONE]".equals(payload)) {
                break;
            }
            JsonNode chunk = mapper.readTree(payload);
            JsonNode delta = chunk.path("choices").path(0).path("delta");

            if (delta.hasNonNull("content")) {
                String token = delta.get("content").asText();
                if (token.length() > 0) {
                    content.append(token);
                    handler.onToken(token);
                }
            }
            for (JsonNode tc : delta.path("tool_calls")) {
                int index = tc.path("index").asInt(0);
                ToolCallBuilder b = builders.get(index);
                if (b == null) {
                    b = new ToolCallBuilder();
                    builders.put(index, b);
                }
                if (tc.hasNonNull("id")) {
                    b.id = tc.get("id").asText();
                }
                JsonNode fn = tc.path("function");
                if (fn.hasNonNull("name")) {
                    b.name.append(fn.get("name").asText());
                }
                if (fn.hasNonNull("arguments")) {
                    b.arguments.append(fn.get("arguments").asText());
                }
            }
        }

        if (builders.isEmpty()) {
            return ChatMessage.assistant(content.toString());
        }
        List<ToolCall> calls = new ArrayList<ToolCall>();
        for (ToolCallBuilder b : builders.values()) {
            String args = b.arguments.length() == 0 ? "{}" : b.arguments.toString();
            calls.add(new ToolCall(b.id, b.name.toString(), args));
        }
        String text = content.length() == 0 ? null : content.toString();
        return ChatMessage.assistantWithToolCalls(text, calls);
    }

    private static final class ToolCallBuilder {
        String id;
        final StringBuilder name = new StringBuilder();
        final StringBuilder arguments = new StringBuilder();
    }

    private String bodyText(Response resp) {
        try {
            return resp.body() == null ? "" : resp.body().string();
        } catch (Exception ex) {
            return "<unreadable>";
        }
    }
}
```
